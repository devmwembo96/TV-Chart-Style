// CX – Pression Achat / Vente · Force & Direction de tendance
// Écriture originale inspirée de « Buying/Selling Pressure & Trend Strength/Direction (Zeiierman) »
// (code d'origine protégé : le calcul ci-dessous est propre à CX, d'après la description publique).
//
// Lecture :
//   • colonnes vertes  = pression acheteuse, rapportée à son propre historique (≈ +200 = normale)
//   • colonnes rouges  = pression vendeuse,  rapportée à son propre historique (≈ −200 = normale)
//   • Fast Trend       = dominance nette rapide (premier signe de changement)
//   • Core Trend       = dominance de fond (> 0 régime haussier, < 0 régime baissier)
//   • Established Trend = ligne épaisse en tirets autour de ±120 : tendance installée
//   • Strong Trend     = points sur le Fast Trend : tendance forte et étirée (risque d'essoufflement)
//   • Long / Short Covered = colonnes jaunes / cyan : débouclage forcé (squeeze, capitulation)
//   • Pressure Cross   = gros points à ±450 · Pre Signals = petits points à ±450

//@version=6
indicator("CX Pression Achat / Vente · Force & Direction", shorttitle = "CX Press", overlay = false, precision = 1, max_labels_count = 50, max_boxes_count = 50)

// ============================================================================
// 1. RÉGLAGES
// ============================================================================
string G_CAL = "Calcul de la pression"
int   candleLen = input.int(14, "Candle Length", minval = 2, maxval = 200, group = G_CAL, tooltip = "Fenêtre de la pression. Au-dessus de 30 : capte les mouvements longs et soutenus.")
int   smoothLen = input.int(3, "Smoothed", minval = 1, maxval = 50, group = G_CAL, tooltip = "Lissage des colonnes. 10 à 30 pour une lecture de tendance plus stable.")
int   normLen   = input.int(200, "Historique de référence", minval = 50, maxval = 1000, group = G_CAL, tooltip = "Chaque côté est comparé à sa propre moyenne sur cette fenêtre : 200 = pression normale.")
bool  useVol    = input.bool(true, "Pondérer par le volume relatif", group = G_CAL, tooltip = "Sans volume (forex), le calcul reste sur le prix seul.")

string G_TRD = "Tendance"
int   fastLen   = input.int(5, "Fast Trend", minval = 1, maxval = 50, group = G_TRD)
int   coreLen   = input.int(50, "Core Trend", minval = 10, maxval = 400, group = G_TRD)
int   estBars   = input.int(8, "Established Trend : bougies d'accord", minval = 2, maxval = 50, group = G_TRD, tooltip = "Fast et Core Trend du même côté depuis au moins N bougies.")
float strongSd  = input.float(1.8, "Strong Trend : écarts-types", minval = 0.5, maxval = 4.0, step = 0.1, group = G_TRD)

string G_SIG = "Signaux"
bool  showCover = input.bool(true, "Long / Short Covered", group = G_SIG)
float coverZ    = input.float(2.5, "Seuil Covered (écarts-types)", minval = 1.0, maxval = 5.0, step = 0.1, group = G_SIG)
bool  showCross = input.bool(true, "Pressure Cross", group = G_SIG)
bool  showPre   = input.bool(true, "Pre Signals", group = G_SIG)

string G_VIS = "Affichage"
bool  showCols  = input.bool(true, "Colonnes de pression", group = G_VIS)
bool  showFast  = input.bool(true, "Fast Trend", group = G_VIS)
bool  showCore  = input.bool(true, "Core Trend", group = G_VIS)
bool  showEst   = input.bool(true, "Established Trend", group = G_VIS)
bool  showStrong = input.bool(true, "Strong Trend", group = G_VIS)
bool  showGauge = input.bool(true, "Jauge de force (à droite)", group = G_VIS)
bool  showTable = input.bool(true, "Tableau Trend / Market State", group = G_VIS)
string tblPosIn = input.string("Top Right", "Position du tableau", options = ["Top Right", "Bottom Right", "Top Left", "Bottom Left"], group = G_VIS)
bool  colorBars = input.bool(false, "Colorer les bougies du graphique", group = G_VIS)
color bullCol   = input.color(#00E676, "Haussier / Baissier", inline = "c", group = G_VIS)
color bearCol   = input.color(#FF1744, "", inline = "c", group = G_VIS)
color lcCol     = input.color(#FFD600, "Long / Short Covered", inline = "cv", group = G_VIS)
color scCol     = input.color(#00E5FF, "", inline = "cv", group = G_VIS)

// ============================================================================
// 2. PRESSION DE CHAQUE CÔTÉ
// ============================================================================
// pression « vraie » : l'écart avec la clôture précédente compte (gaps inclus)
float prevC   = nz(close[1], open)
float buyRaw  = close - math.min(low, prevC)
float sellRaw = math.max(high, prevC) - close

float volMa   = ta.sma(volume, candleLen)
float relVol  = useVol and nz(volume) > 0 and nz(volMa) > 0 ? math.min(volume / volMa, 5.0) : 1.0

float buyE    = ta.ema(buyRaw * relVol, candleLen)
float sellE   = ta.ema(sellRaw * relVol, candleLen)
float buyBase = ta.sma(buyE, normLen)
float sellBase = ta.sma(sellE, normLen)

// chaque côté rapporté à sa propre normale : 200 = pression habituelle
float buyP  = ta.ema(nz(buyBase) > 0 ? 200.0 * buyE / buyBase : 0.0, smoothLen)
float sellP = -ta.ema(nz(sellBase) > 0 ? 200.0 * sellE / sellBase : 0.0, smoothLen)

// dominance nette (acheteurs − vendeurs), amplifiée pour la lecture
float netP  = (buyP + sellP) * 2.0
float fastT = ta.ema(netP, fastLen)
float coreT = ta.ema(netP, coreLen)

// ============================================================================
// 3. FORCE, ÉTAT DE TENDANCE
// ============================================================================
float fastMean = ta.sma(fastT, normLen)
float fastSd   = ta.stdev(fastT, normLen)
bool  strongUp = fastT > 0 and coreT > 0 and fastT > fastMean + strongSd * fastSd
bool  strongDn = fastT < 0 and coreT < 0 and fastT < fastMean - strongSd * fastSd

int   upRun    = ta.barssince(not (fastT > 0 and coreT > 0))
int   dnRun    = ta.barssince(not (fastT < 0 and coreT < 0))
bool  estUp    = nz(upRun) >= estBars
bool  estDn    = nz(dnRun) >= estBars

// force relative de chaque côté (rang sur l'historique récent)
float bullStr  = ta.percentrank(buyP, 100)
float bearStr  = ta.percentrank(-sellP, 100)

// ============================================================================
// 4. SIGNAUX
// ============================================================================
// Long / Short Covered : anomalie de la pression de la bougie contre le régime de fond
float barNet   = (buyRaw - sellRaw) * relVol
float barMa    = ta.sma(barNet, 50)
float barSd    = ta.stdev(barNet, 50)
float barZ     = nz(barSd) > 0 ? (barNet - barMa) / barSd : 0.0
bool  shortCov = barZ >= coverZ and coreT < 0       // vendeurs forcés de racheter
bool  longCov  = barZ <= -coverZ and coreT > 0      // acheteurs forcés de vendre

// Pressure Cross : un côté passe devant l'autre
float buyAbs   = buyP
float sellAbs  = -sellP
bool  crossUp  = ta.crossover(buyAbs, sellAbs)
bool  crossDn  = ta.crossunder(buyAbs, sellAbs)

// Pre Signals : le Fast Trend se retourne avant que la dominance ne change de camp
float fastSig  = ta.ema(fastT, 5)
bool  preUp    = ta.crossover(fastT, fastSig) and fastT < 0 and not crossUp
bool  preDn    = ta.crossunder(fastT, fastSig) and fastT > 0 and not crossDn

// ============================================================================
// 5. AFFICHAGE
// ============================================================================
hline(0, "Zéro", color = color.new(color.gray, 50), linestyle = hline.style_solid)
hline(400, "+400", color = color.new(color.gray, 75), linestyle = hline.style_dashed)
hline(-400, "−400", color = color.new(color.gray, 75), linestyle = hline.style_dashed)

// colonnes de pression : intensité selon la force relative
color buyColN  = color.new(bullCol, math.round(80 - 70 * nz(bullStr) / 100))
color sellColN = color.new(bearCol, math.round(80 - 70 * nz(bearStr) / 100))
color buyColF  = shortCov and showCover ? scCol : buyColN
color sellColF = longCov and showCover ? lcCol : sellColN
plot(showCols ? buyP : na,  "Pression acheteuse", color = buyColF,  style = plot.style_columns)
plot(showCols ? sellP : na, "Pression vendeuse",  color = sellColF, style = plot.style_columns)

// Core Trend (fond) et Fast Trend (rapide)
color coreCol = coreT >= 0 ? color.new(bullCol, 40) : color.new(bearCol, 40)
plot(showCore ? coreT : na, "Core Trend", color = coreCol, linewidth = 1)
color fastCol = fastT >= fastT[1] ? bullCol : bearCol
plot(showFast ? fastT : na, "Fast Trend (halo)", color = color.new(fastCol, 80), linewidth = 6)
plot(showFast ? fastT : na, "Fast Trend", color = fastCol, linewidth = 2)

// Strong Trend : points sur le Fast Trend
plot(showStrong and (strongUp or strongDn) ? fastT : na, "Strong Trend", color = strongUp ? bullCol : bearCol, style = plot.style_circles, linewidth = 4)

// Established Trend : ligne épaisse en tirets autour de ±120
plot(showEst and estUp ? 120 : na, "Established Trend haussier", color = bullCol, style = plot.style_cross, linewidth = 4)
plot(showEst and estDn ? -120 : na, "Established Trend baissier", color = bearCol, style = plot.style_cross, linewidth = 4)

// Pressure Cross (gros points) et Pre Signals (petits points) à ±450
plot(showCross and crossUp ? 450 : na, "Pressure Cross haussier", color = bullCol, style = plot.style_circles, linewidth = 5)
plot(showCross and crossDn ? -450 : na, "Pressure Cross baissier", color = bearCol, style = plot.style_circles, linewidth = 5)
plot(showPre and preUp ? 450 : na, "Pre Signal haussier", color = color.new(bullCol, 45), style = plot.style_circles, linewidth = 2)
plot(showPre and preDn ? -450 : na, "Pre Signal baissier", color = color.new(bearCol, 45), style = plot.style_circles, linewidth = 2)

// Long / Short Covered : repère au-dessus / au-dessous des colonnes
plotshape(showCover and shortCov, "Short Covered", shape.diamond, location.bottom, scCol, size = size.tiny)
plotshape(showCover and longCov,  "Long Covered",  shape.diamond, location.top,    lcCol, size = size.tiny)

// couleur des bougies du graphique principal (option)
barcolor(colorBars ? (netP >= 0 ? color.new(bullCol, math.round(70 - 60 * nz(bullStr) / 100)) : color.new(bearCol, math.round(70 - 60 * nz(bearStr) / 100))) : na, title = "Bougies selon la pression")

// ============================================================================
// 6. JAUGE DE FORCE ET TABLEAU
// ============================================================================
var box   gBull = na
var box   gBear = na
var label lBull = na
var label lBear = na

if barstate.islast
    box.delete(gBull)
    box.delete(gBear)
    label.delete(lBull)
    label.delete(lBear)
    if showGauge
        float bs = nz(bullStr)
        float rs = nz(bearStr)
        gBull := box.new(bar_index + 3, 4.0 * bs, bar_index + 4, 0, border_color = bullCol, bgcolor = color.new(bullCol, 30))
        gBear := box.new(bar_index + 3, 0, bar_index + 4, -4.0 * rs, border_color = bearCol, bgcolor = color.new(bearCol, 30))
        lBull := label.new(bar_index + 5, 4.0 * bs, str.tostring(bs, "#") + "% Bullish Strength", style = label.style_label_left, color = color.new(color.black, 100), textcolor = bullCol, size = size.small)
        lBear := label.new(bar_index + 5, -4.0 * rs, str.tostring(rs, "#") + "% Bearish Strength", style = label.style_label_left, color = color.new(color.black, 100), textcolor = bearCol, size = size.small)

string tblPos = tblPosIn == "Bottom Right" ? position.bottom_right : tblPosIn == "Top Left" ? position.top_left : tblPosIn == "Bottom Left" ? position.bottom_left : position.top_right
var table tbl = table.new(tblPos, 2, 2, border_width = 0)

if barstate.islast
    table.clear(tbl, 0, 0, 1, 1)
    if showTable
        string trendTxt = coreT > 0 ? "Bullish" : coreT < 0 ? "Bearish" : "Neutral"
        color  trendBg  = coreT > 0 ? bullCol : coreT < 0 ? bearCol : color.gray
        string stateTxt = strongUp or strongDn ? "Strong · Stretched" : estUp or estDn ? "Trending" : math.abs(coreT) < 40 ? "Ranging" : "Building"
        color  stateBg  = strongUp or strongDn ? color.orange : estUp or estDn ? trendBg : color.gray
        table.cell(tbl, 0, 0, "Trend", text_color = color.white, bgcolor = #2962FF, text_size = size.small)
        table.cell(tbl, 1, 0, "Market State", text_color = color.white, bgcolor = #2962FF, text_size = size.small)
        table.cell(tbl, 0, 1, trendTxt, text_color = color.white, bgcolor = trendBg, text_size = size.small)
        table.cell(tbl, 1, 1, stateTxt, text_color = color.white, bgcolor = stateBg, text_size = size.small)

// ============================================================================
// 7. ALERTES
// ============================================================================
bool estUpStart    = estUp and not estUp[1]
bool estDnStart    = estDn and not estDn[1]
bool coreUpTurn    = coreT > 0 and coreT[1] <= 0
bool coreDnTurn    = coreT < 0 and coreT[1] >= 0

alertcondition(crossUp,    "Pressure Cross haussier", "CX Pression : les acheteurs passent devant sur {{ticker}} ({{interval}})")
alertcondition(crossDn,    "Pressure Cross baissier", "CX Pression : les vendeurs passent devant sur {{ticker}} ({{interval}})")
alertcondition(preUp,      "Pre Signal haussier", "CX Pression : premier signe de pression acheteuse sur {{ticker}} ({{interval}})")
alertcondition(preDn,      "Pre Signal baissier", "CX Pression : premier signe de pression vendeuse sur {{ticker}} ({{interval}})")
alertcondition(shortCov,   "Short Covered", "CX Pression : Short Covered (rachat forcé des vendeurs) sur {{ticker}} ({{interval}})")
alertcondition(longCov,    "Long Covered", "CX Pression : Long Covered (vente forcée des acheteurs) sur {{ticker}} ({{interval}})")
alertcondition(estUpStart, "Established Trend haussier", "CX Pression : tendance haussière installée sur {{ticker}} ({{interval}})")
alertcondition(estDnStart, "Established Trend baissier", "CX Pression : tendance baissière installée sur {{ticker}} ({{interval}})")
alertcondition(coreUpTurn, "Core Trend passe haussier", "CX Pression : le régime de fond passe haussier sur {{ticker}} ({{interval}})")
alertcondition(coreDnTurn, "Core Trend passe baissier", "CX Pression : le régime de fond passe baissier sur {{ticker}} ({{interval}})")
