f_trendMajor(int len) =>
    float hs = lineSw ? close : math.max(open, close)
    float ls = lineSw ? close : math.min(open, close)
    float pH = ta.pivothigh(hs, len, len)
    float pL = ta.pivotlow(ls, len, len)
    float loK = ta.lowest(ls, len)
    float hiK = ta.highest(hs, len)
    var int   tr  = 0
    var float lPH = na
    var float lPL = na
    var float xH  = na
    var float xL  = na
    var float pHi = na
    var float pLo = na
    var float oLo = na
    var float oHi = na
    // origin of the current move
    if not na(pH)
        oLo := loK
    else if na(oLo) or ls < oLo
        oLo := ls
    if not na(pL)
        oHi := hiK
    else if na(oHi) or hs > oHi
        oHi := hs
    if not na(pH)
        lPH := pH
        if na(xH) or pH > xH
            xH := pH
    if not na(pL)
        lPL := pL
        if na(xL) or pL < xL
            xL := pL
    int e = 0
    if tr == 1
        if not na(pLo) and close < pLo
            e := -1
        else if not na(xH) and close > xH
            e := 1
    else if tr == -1

        xL  := na
    if e < 0

        pHi := na(oHi) ? lPH : oHi

f_eq(float tolM) =>
    float a  = ta.atr(14)
    float ph = ta.pivothigh(high, 2, 2)
    float pl = ta.pivotlow(low, 2, 2)
    var array<float> hp  = array.new<float>()
    var array<int>   htm = array.new<int>()
    var array<float> lq  = array.new<float>()
    var array<int>   ltm = array.new<int>()
    var float eqH  = na
    var int   eqHt = na
    var float eqL  = na
    var int   eqLt = na
    if hp.size() > 0
        for i = hp.size() - 1 to 0
            if high > hp.get(i)
                hp.remove(i)
                htm.remove(i)
    if lq.size() > 0
        for i = lq.size() - 1 to 0
            if low < lq.get(i)
                lq.remove(i)
                ltm.remove(i)
    if not na(eqH) and high > eqH
        eqH := na
    if not na(eqL) and low < eqL
        eqL := na
    if not na(ph)
        int pt = time[2]
        if hp.size() > 0
            for i = 0 to hp.size() - 1
                if math.abs(ph - hp.get(i)) <= tolM * a and pt - htm.get(i) <= 86400000
                    eqH  := math.max(ph, hp.get(i))
                    eqHt := pt
        hp.push(ph)
        htm.push(pt)
        if hp.size() > 8
            hp.shift()
            htm.shift()
    if not na(pl)
        int pt = time[2]
        if lq.size() > 0
            for i = 0 to lq.size() - 1
                if math.abs(pl - lq.get(i)) <= tolM * a and pt - ltm.get(i) <= 86400000
                    eqL  := math.min(pl, lq.get(i))
                    eqLt := pt
        lq.push(pl)
        ltm.push(pt)
        if lq.size() > 8
            lq.shift()
            ltm.shift()
    if not na(eqHt) and time - eqHt > 86400000
        eqH := na
    if not na(eqLt) and time - eqLt > 86400000
        eqL := na
    [eqH, eqL]

f_eval(array<float> lvP, array<string> lvN, array<bool> lvH, int dir, float nearP, float farP, float cd) =>
    array<float>  cp = array.new<float>()
    array<string> cn = array.new<string>()
    float lo = math.min(nearP, farP)
    float hi = math.max(nearP, farP)
    bool wantH = dir == -1
    if lvP.size() > 0
        for i = 0 to lvP.size() - 1
            float p = lvP.get(i)
            if lvH.get(i) == wantH and p > lo and p < hi
                cp.push(p)
                cn.push(lvN.get(i))
    int tot = cp.size()
    int best = 0
    string bn = ""
    float bd = na
    if tot > 0
        for i = 0 to tot - 1
            float base = cp.get(i)
            int c = 0
            string nm = ""
            float mx = base
            for j = 0 to tot - 1
                float pj = cp.get(j)
                if pj >= base and pj - base <= cd
                    c += 1
                    nm := nm == "" ? cn.get(j) : nm + " + " + cn.get(j)
                    mx := math.max(mx, pj)
            float d = math.abs(farP - (base + mx) / 2)
            if c > best or (c == best and (na(bd) or d < bd))
                best := c
                bn   := nm
                bd   := d
    [best, tot, bn]

// CheckMitigated : comblée selon le réglage (sur les bougies du graphique)
method CheckMitigated(ImbalanceStructure IS, float o, float h, float l, float c) =>
    if IS.imbalance.size() > 0
        for i = IS.imbalance.size() - 1 to 0
            Imbalance imb = IS.imbalance.get(i)
            if not imb.mitigated
                switch mit_type
                    "None" =>
                        imb.mitigated := false
                    "Wick filled" =>
                        imb.mitigated := imb.open <= imb.close ? low <= imb.open : high >= imb.open
                    "Body filled" =>
                        imb.mitigated := imb.open < imb.close ? math.min(o, c) <= imb.open : math.max(o, c) >= imb.open
                    "Wick filled half" =>
                        imb.mitigated := imb.open <= imb.close ? low <= imb.middle : high >= imb.middle
                    "Body filled half" =>
                        imb.mitigated := imb.open <= imb.close ? math.min(o, c) <= imb.middle : math.max(o, c) >= imb.middle
                if imb.mitigated
                    if not mit_show
                        f_imbDel(imb)
                        IS.imbalance.remove(i)
                    else
                        imb.mitigated_time := time
    IS

// AdjustMargins : au plus max_count boîtes par unité, à moins d'une ATR Daily du prix
method AdjustMargins(ImbalanceStructure IS, int step) =>
    int count = 0
    if IS.imbalance.size() > 0
        float rl = timeframe.isintraday ? close - nz(dATR) : 0
        float rh = timeframe.isintraday ? close + nz(dATR) : 0
        for i = 0 to IS.imbalance.size() - 1
            Imbalance imb = IS.imbalance.get(i)
            if fvgDraw and cxOn and (math.max(imb.open, imb.close) > rl) and (math.min(imb.open, imb.close) < rh) and count < IS.settings.max_count
                if IS.AddZone(imb, step)
                    count := count + 1
            else
                f_imbDel(imb)
    IS

// TP : premier niveau de liquidité vivant dans le sens du trade qui donne au moins le R
// minimum ; sinon R fixe.
f_tp(int dir, float en, float sl) =>
    float risk = math.abs(en - sl)
    float tp = na
    string nm = ""
    if tpMode == "Liquidité" and risk > 0 and lvP.size() > 0
        for i = 0 to lvP.size() - 1
            float p = lvP.get(i)
            if (dir == 1 ? p > en : p < en) and math.abs(p - en) / risk >= minR
                if na(tp) or math.abs(p - en) < math.abs(tp - en)
                    tp := p
                    nm := lvN.get(i)
    if na(tp) and risk > 0
        tp := en + dir * fixR * risk
        nm := str.tostring(fixR, "#.#") + "R fixe"
    [tp, risk > 0 ? math.abs(tp - en) / risk : na, nm]

f_filt(bool uB, bool uT, bool uP, int dir, float rr) =>
    bool b = not uB or tradeB == dir
    bool t = not uT or inTrdX
    bool p = not uP or (dir == 1 ? pdZ == -1 : pdZ == 1)
    bool r = not useRR or (not na(rr) and rr >= minR)
    b and t and p and r

    tr.bS := box.new(bar_index, math.max(en, sl), x2, math.min(en, sl), border_color = color(na), bgcolor = slZone)
    tr.bT := box.new(bar_index, math.max(en, tp), x2, math.min(en, tp), border_color = color(na), bgcolor = tpZone)

    if not na(cy) and not na(cx1)

    tr

// suivi : TP touché → ✓ TP, SL touché → ✕ SL (même bougie : SL compté)
f_upd(array<Trd> H) =>
    if H.size() > 0
        for tr in H
            if tr.st == 0
                if bar_index > tr.bar
                    bool hS = tr.dir == 1 ? low <= tr.sl : high >= tr.sl
                    bool hT = tr.dir == 1 ? high >= tr.tp : low <= tr.tp
                    if hS or hT
                        tr.st   := hS ? 2 : 1
                        tr.rbar := bar_index
                        label.set_text(tr.lb, tr.txt + (hS ? " ✕ SL" : " ✓ TP"))
                // largeur fixe ; une résolution plus tôt arrête les zones sur sa bougie
                int x2 = tr.st == 0 ? tr.bar + tradeBars : math.min(tr.rbar, tr.bar + tradeBars)
                line.set_x2(tr.lE, x2)
                line.set_x2(tr.lS, x2)
                line.set_x2(tr.lT, x2)
                box.set_right(tr.bS, x2)
                box.set_right(tr.bT, x2)
                label.set_x(tr.lb, x2)
    true

f_zUpd(Zt z) =>
    bool inNow = z.dir == 1 ? low <= z.hi : high >= z.lo
    bool newC = false
    if inNow and not z.inPrev and bar_index > z.born
        z.touches += 1
        z.cBar    := bar_index
        z.ext     := z.dir == 1 ? low : high
        z.extBar  := bar_index
        z.done1   := false
        z.done3   := false
        newC      := true
    else if not na(z.cBar) and bar_index - z.cBar <= s2Wait
        if z.dir == 1 ? low < z.ext : high > z.ext
            z.ext    := z.dir == 1 ? low : high
            z.extBar := bar_index
    if z.dir == 1 ? close < z.lo : close > z.hi
        z.dead := true
    z.inPrev := inNow
    newC

// rejet « mèche » : clôture hors de la zone côté réaction, mèche dans la zone ≥ X % de la bougie
f_rejW(Zt z) =>
    float rg = high - low
    rg > 0 and (z.dir == 1 ? close > z.hi and low <= z.hi and (math.min(z.hi, math.min(open, close)) - math.max(z.lo, low)) / rg * 100 >= s1Wick : close < z.lo and high >= z.lo and (math.min(z.hi, high) - math.max(z.lo, math.max(open, close))) / rg * 100 >= s1Wick)

// rejet « englobante » : le corps englobe celui de la bougie précédente et clôture hors de la zone
f_rejE(Zt z) =>
    z.dir == 1 ? close > open and close > z.hi and close > math.max(open[1], close[1]) and open <= math.min(open[1], close[1]) and (low <= z.hi or low[1] <= z.hi) : close < open and close < z.lo and close < math.min(open[1], close[1]) and open >= math.max(open[1], close[1]) and (high >= z.lo or high[1] >= z.lo)

// CISD : ouverture de la première bougie de la série de bougies opposées qui a fait
// l'extrême (achat : série baissière qui a fait le plus bas).
f_cisdLvl(int dir, int extBar) =>
    float lvl = na
    int   lb  = na
    if not na(extBar)
        int off = bar_index - extBar
        if off >= 0 and off < 400
            int k = off
            bool opp = dir == 1 ? close[k] < open[k] : close[k] > open[k]
            if not opp
                k += 1
            int n = 0
            while n < 30
                bool o2 = dir == 1 ? close[k] < open[k] : close[k] > open[k]
                if not o2
                    break
                lvl := open[k]
                lb  := bar_index - k
                k   += 1
                n   += 1
    [lvl, lb]

f_cisdHit(int dir, int extBar) =>
    [lvl, lb] = f_cisdLvl(dir, extBar)
    bool hit = not na(lvl) and (dir == 1 ? close > lvl and close[1] <= lvl : close < lvl and close[1] >= lvl)
    [hit, lvl, lb]

f_win(string mode) =>
    mode == "CISD" ? s2Wait : s1Wait

// confirmation d'une zone touchée : mèche, englobante ou CISD
f_conf(string mode, Zt z) =>
    bool ok = false
    string tx = ""
    float cy = na
    int cx = na
    bool inW = not na(z.cBar) and bar_index - z.cBar <= f_win(mode)
    [h, l, b] = f_cisdHit(z.dir, z.extBar)
    if mode == "Mèche"
        ok := inW and f_rejW(z)
        tx := "rejet mèche"
    else if mode == "Englobante"
        ok := inW and f_rejE(z)
        tx := "rejet englobante"
    else
        ok := inW and h
        cy := l
        cx := b
        tx := "CISD " + (na(l) ? "—" : f_px(l))
    [ok, tx, cy, cx]

atr_multi = input.float(0.25, step = 0.25,minval = 0,  title = "ATR Multiplier", tooltip = "Filters FVGs based on ATR Width, Only displays Inversions that are Greater-Than the ATR*Multiplier.")

//Colors
green = input.color(color.new(#089981, 80), title = "Bull Color", group = "Colors")
red = input.color(color.new(#f23645, 80), title = "Bear Color", group = "Colors")
gray = input.color(#787b86, title = "Midline Color", group = "Colors")
invis = color.rgb(0,0,0,100)

//---------------------------------------------------------------------------------------------------------------------}
//UDT's
//---------------------------------------------------------------------------------------------------------------------{
type lab //Contains Necessary Label Data to Send to Label Function
    int x
    float y
    int dir

type fvg //Contains Necessary FVG Data to Send to Chart.
    int left = na
    float top = na
    int right = na
    float bot = na
    float mid = na
    int dir = na
    int state = na
    array<lab> labs = na
    int x_val = na

//---------------------------------------------------------------------------------------------------------------------}
//Functions
//---------------------------------------------------------------------------------------------------------------------{
//Basic Calcs
buffer = 100 //How many FVGs to keep in memory.
c_top = math.max(open,close)
c_bot = math.min(open,close)

label_maker(_x,_y,_dir) => //Used for making Labels
    switch
        _dir == 1 => ifvgLabels.push(label.new(_x,_y,"\n▲", style = label.style_text_outline, color = invis,  textcolor = color.new(green,0), size = size.small, xloc = xloc.bar_time))
        _dir == -1 => ifvgLabels.push(label.new(_x,_y, "▼\n", style = label.style_text_outline, color = invis,  textcolor = color.new(red,0), size = size.small, xloc = xloc.bar_time))

fvg_manage(_ary,_inv_ary) => //First step filtering of FVG data, Not all FVGs will be displayed, only inversions.
    if _ary.size() >= buffer
        _ary.shift()

    if _ary.size() > 0
        for i = _ary.size()-1 to 0
            value = _ary.get(i)
            _dir = value.dir

            if _dir == 1 and (c_bot < value.bot)
                value.x_val := time
                _inv_ary.push(_ary.remove(i))
            if _dir == -1 and (c_top > value.top)
                value.x_val := time
                _inv_ary.push(_ary.remove(i))
            

inv_manage(_ary) => //All inversions will be displayed.
    fire = false
    if _ary.size() >= buffer
        _ary.shift()
    if _ary.size() > 0
        for i = _ary.size()-1 to 0
            value = _ary.get(i)
            bx_top = value.top
            bx_bot = value.bot
            _dir = value.dir
            st = value.state

            if (st == 0 and _dir == 1)
                value.state := 1
                value.dir := -1
            if (_dir == -1 and st == 0)
                value.state := 1
                value.dir := 1
            if st >= 1
                value.right := time
            if (_dir == -1 and st == 1 and close < bx_bot and (wt?high:close[1]) >= bx_bot and (wt?high:close[1]) < bx_top)
                value.labs.push(lab.new(time,bx_top,-1))
                fire := true
            if (_dir == 1 and st == 1 and close > bx_top and (wt?low:close[1]) <= bx_top and (wt?low:close[1]) > bx_bot)
                value.labs.push(lab.new(time,bx_bot,1))
                fire := true
            if st >= 1 and ((_dir == -1 and c_top > bx_top) or (_dir == 1 and c_bot < bx_bot))
                _ary.remove(i)
            
    fire

send_it(_ary) => // Draws Everything on the Chart
    last_index = _ary.size()-1
    for [index,value] in _ary
        bx_top = value.top
        bx_bot = value.bot
        bx_left = value.left
        xval = value.x_val
        mid = value.mid
        col = value.dir == -1 ? green : red
        o_col = value.dir == -1 ? red : green

        if index > last_index - disp_num
            ifvgBoxes.push(box.new(bx_left,bx_top,xval,bx_bot,bgcolor = col, border_color = invis, xloc = xloc.bar_time))
            ifvgBoxes.push(box.new(xval,bx_top,time,bx_bot, bgcolor = o_col, border_color = invis, xloc = xloc.bar_time))

            ifvgLines.push(line.new(bx_left,mid,time,mid, color = gray, style = line.style_dashed, xloc = xloc.bar_time))
            ifvgBoxes.push(box.new(bar_index,bx_top,bar_index+50,bx_bot, bgcolor = o_col, border_color = invis))
            ifvgLines.push(line.new(bar_index,mid,bar_index+50,mid, color = gray, style = line.style_dashed))

            for stuff in value.labs
                label_maker(stuff.x,stuff.y,stuff.dir)

//---------------------------------------------------------------------------------------------------------------------}
//Delete drawings
//---------------------------------------------------------------------------------------------------------------------{
// [CX] L'original efface ici tous les dessins du script à chaque bougie. Dans CX Liquidity,
// cela effacerait aussi les séances, niveaux et POI : seuls les dessins IFVG sont effacés.
for boxes in ifvgBoxes
    box.delete(boxes)
for lines in ifvgLines
    line.delete(lines)
for labels in ifvgLabels
    label.delete(labels)
ifvgBoxes.clear()
ifvgLines.clear()
ifvgLabels.clear()

//---------------------------------------------------------------------------------------------------------------------}
//Data Arrays
//---------------------------------------------------------------------------------------------------------------------{
var bull_fvg_ary = array.new<fvg>() // FVG Data, Not all will be Drawn
var bear_fvg_ary = array.new<fvg>()

var bull_inv_ary = array.new<fvg>() // Inversion Data, All will be Drawn
var bear_inv_ary = array.new<fvg>()

//---------------------------------------------------------------------------------------------------------------------}
//FVG Detection
//---------------------------------------------------------------------------------------------------------------------{
atr = nz(ta.atr(200)*atr_multi, ta.cum(high - low) / (bar_index+1))

fvg_up = (low > high[2]) and (close[1] > high[2])
fvg_down = (high < low[2]) and (close[1] < low[2])

if fvg_up and math.abs(low-high[2]) > atr
    array.push(bull_fvg_ary,fvg.new(time[1], low, time, high[2], math.avg(low,high[2]), 1, 0,array.new<lab>(),na))

if fvg_down and math.abs(low[2]-high) > atr
    array.push(bear_fvg_ary,fvg.new(time[1], low[2], time, high, math.avg(high,low[2]),-1 ,0,array.new<lab>(),na))

//---------------------------------------------------------------------------------------------------------------------}
//Running Functions
//---------------------------------------------------------------------------------------------------------------------{
// FVG_Data -> Inversion_Data -> Chart
fvg_manage(bull_fvg_ary,bull_inv_ary)
fvg_manage(bear_fvg_ary,bear_inv_ary)

ifvg_bear_signal = inv_manage(bull_inv_ary)
ifvg_bull_signal = inv_manage(bear_inv_ary)

if barstate.islast
    send_it(bull_inv_ary)
    send_it(bear_inv_ary)

alertcondition(ifvg_bull_signal, "IFVG Bullish Signal")
alertcondition(ifvg_bear_signal, "IFVG Bearish Signal")
