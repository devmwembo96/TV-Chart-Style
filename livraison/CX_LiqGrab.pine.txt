f_autoLtf() =>
    int s = timeframe.in_seconds()
    s <= 900 ? "1" : s <= 3600 ? "5" : s <= 14400 ? "15" : s <= 86400 ? "60" : "240"

string ltf = ltfIn == "Auto" ? f_autoLtf() : ltfIn

type Sfp
    int   dir      // 1 = haussier (liquidité basse prise), -1 = baissier
    float lvl      // niveau du swing pris
    float ext      // extrême de la mèche du SFP
    line  ln
    box   bx
    label lb
    bool  alive

var array<Sfp> sfps = array.new<Sfp>()
var array<box>   fvgs   = array.new<box>()
var array<int>   fvgDir = array.new<int>()
var array<float> fvgTop = array.new<float>()
var array<float> fvgBot = array.new<float>()
bool inWin = bar_index >= last_bar_index - drawBars

// ============================================================================
// 3. VOLUME DE L'UT INFÉRIEURE
// ============================================================================
[ltfH, ltfL, ltfV] = request.security_lower_tf(syminfo.tickerid, ltf, [high, low, volume], calc_bars_count = useVal ? 20000 : 10)

// part (%) du volume de la bougie traitée au-delà d'un niveau (dir 1 : sous le niveau, -1 : au-dessus)
f_outVolPct(float lvl, int dir) =>
    float tot = 0.0
    float out = 0.0
    int n = array.size(ltfV)
    if n > 0
        for i = 0 to n - 1
            float v = nz(array.get(ltfV, i))
            tot += v
            if dir == 1 ? array.get(ltfL, i) < lvl : array.get(ltfH, i) > lvl
                out += v
    tot > 0 ? out / tot * 100.0 : na

// ============================================================================
// 4. SWING FAILURE PATTERN (prise de liquidité)
// ============================================================================
float swH = ta.pivothigh(high, sfpLen, sfpLen)
float swL = ta.pivotlow(low, sfpLen, sfpLen)

var float lvlH = na
var int   lvlHB = na
var float lvlL = na
var int   lvlLB = na

if not na(swH)
    lvlH  := swH
    lvlHB := bar_index - sfpLen
if not na(swL)
    lvlL  := swL
    lvlLB := bar_index - sfpLen

bool rawBearSfp = not na(lvlH) and high > lvlH and close < lvlH and open < lvlH
bool rawBullSfp = not na(lvlL) and low < lvlL and close > lvlL and open > lvlL

float bearVolPct = rawBearSfp ? f_outVolPct(lvlH, -1) : na
float bullVolPct = rawBullSfp ? f_outVolPct(lvlL, 1) : na
bool  bearValid  = not useVal or na(bearVolPct) or bearVolPct >= valThr
bool  bullValid  = not useVal or na(bullVolPct) or bullVolPct >= valThr

bool bearSfp = rawBearSfp and bearValid
bool bullSfp = rawBullSfp and bullValid

var float lastVolPct = na
var int   lastSfpDir = 0

f_newSfp(int dir, float lvl, int lvlBar, float ext) =>
    color c  = dir == 1 ? bullCol : bearCol
    color wc = dir == 1 ? bullWick : bearWick
    line  l  = line.new(lvlBar, lvl, bar_index, lvl, color = c, width = 1)
    box   b  = box.new(bar_index - 1, math.max(lvl, ext), bar_index + 1, math.min(lvl, ext), border_color = c, border_style = line.style_dotted, bgcolor = wc)
    label t  = showLbl ? label.new(bar_index, ext, "✕", style = label.style_none, textcolor = c, size = size.small, yloc = dir == 1 ? yloc.belowbar : yloc.abovebar) : na
    array.push(sfps, Sfp.new(dir, lvl, ext, l, b, t, true))
    if array.size(sfps) > 40
        Sfp o = array.shift(sfps)
        line.delete(o.ln)
        box.delete(o.bx)
        label.delete(o.lb)

if showSfp and inWin and bearSfp
    f_newSfp(-1, lvlH, lvlHB, high)
if showSfp and inWin and bullSfp
    f_newSfp(1, lvlL, lvlLB, low)
if bearSfp
    lastVolPct := bearVolPct
    lastSfpDir := -1
    lvlH := na
if bullSfp
    lastVolPct := bullVolPct
    lastSfpDir := 1
    lvlL := na
// clôture au-delà du swing : vraie cassure, la liquidité est consommée
if not na(lvlH) and close > lvlH
    lvlH := na
if not na(lvlL) and close < lvlL
    lvlL := na

// invalidation : clôture au-delà de la mèche du SFP
// (seuls les SFP encore valides restent dans la liste : boucle courte)
if array.size(sfps) > 0
    for i = array.size(sfps) - 1 to 0
        Sfp s = array.get(sfps, i)
        if s.dir == 1 ? close < s.ext : close > s.ext
            s.alive := false
            array.remove(sfps, i)
            if sfpHist
                color fc = color.new(s.dir == 1 ? bullCol : bearCol, 75)
                line.set_color(s.ln, fc)
                box.set_border_color(s.bx, fc)
                box.set_bgcolor(s.bx, color.new(fc, 95))
                label.set_textcolor(s.lb, fc)
            else
                line.delete(s.ln)
                box.delete(s.bx)
                label.delete(s.lb)

// ============================================================================
// 5. FAIR VALUE GAPS
// ============================================================================
bool bullFvg = low > high[2] and close[1] > high[2]
bool bearFvg = high < low[2] and close[1] < low[2]

if showFvg and inWin and bullFvg
    array.push(fvgTop, low)
    array.push(fvgBot, high[2])
    array.push(fvgs, box.new(bar_index - 2, low, bar_index, high[2], border_color = na, bgcolor = fvgBullC, extend = extend.right, text = showFvgLb ? "FVG" : "", text_color = color.new(bullCol, 30), text_size = size.tiny, text_halign = text.align_right))
    array.push(fvgDir, 1)
if showFvg and inWin and bearFvg
    array.push(fvgTop, low[2])
    array.push(fvgBot, high)
    array.push(fvgs, box.new(bar_index - 2, low[2], bar_index, high, border_color = na, bgcolor = fvgBearC, extend = extend.right, text = showFvgLb ? "FVG" : "", text_color = color.new(bearCol, 30), text_size = size.tiny, text_halign = text.align_right))
    array.push(fvgDir, -1)

// remplissage : le FVG disparaît quand X % de sa hauteur est comblé
if array.size(fvgs) > 0
    for i = array.size(fvgs) - 1 to 0
        int   d   = array.get(fvgDir, i)
        float tp  = array.get(fvgTop, i)
        float bt  = array.get(fvgBot, i)
        float hgt = math.max(tp - bt, syminfo.mintick)
        bool newOne = i == array.size(fvgs) - 1 and (bullFvg or bearFvg)
        float fillPct = d == 1 ? (tp - low) / hgt * 100.0 : (high - bt) / hgt * 100.0
        if not newOne and fillPct >= fvgInv
            box.delete(array.get(fvgs, i))
            array.remove(fvgs, i)
            array.remove(fvgDir, i)
            array.remove(fvgTop, i)
            array.remove(fvgBot, i)
if array.size(fvgs) > 40
    box.delete(array.shift(fvgs))
    array.shift(fvgDir)
    array.shift(fvgTop)
    array.shift(fvgBot)

// ============================================================================
// 6. STRUCTURE : FRACTALES, BOS / MSS
// ============================================================================
float frH = ta.pivothigh(high, pivLen, pivLen)
float frL = ta.pivotlow(low, pivLen, pivLen)

plotshape(showFrac and not na(frH), "Fractal haut", shape.triangledown, location.abovebar, fracUpCol, offset = -pivLen, size = size.tiny)
plotshape(showFrac and not na(frL), "Fractal bas", shape.triangleup, location.belowbar, fracDnCol, offset = -pivLen, size = size.tiny)

var float msH = na
var int   msHB = na
var float msL = na
var int   msLB = na
var int   trend = 0
var line  lastBullLn = na
var label lastBullLb = na
var line  lastBearLn = na
var label lastBearLb = na

if not na(frH)
    msH  := frH
    msHB := bar_index - pivLen
if not na(frL)
    msL  := frL
    msLB := bar_index - pivLen

bool bullBreak = not na(msH) and close > msH
bool bearBreak = not na(msL) and close < msL
bool bullMss = bullBreak and trend == -1
bool bearMss = bearBreak and trend == 1
bool bullBos = bullBreak and not bullMss
bool bearBos = bearBreak and not bearMss

f_ms(int fromBar, float lvl, string tag, color c, bool up) =>
    line  l = line.new(fromBar, lvl, bar_index, lvl, color = c, style = msStyle, width = 1)
    label t = showLbl ? label.new(math.round((fromBar + bar_index) / 2), lvl, tag, style = up ? label.style_label_down : label.style_label_up, color = color.new(c, 100), textcolor = c, size = size.tiny) : na
    [l, t]

if bullBreak
    if showMs
        if not msHist
            line.delete(lastBullLn)
            label.delete(lastBullLb)
        [l, t] = f_ms(msHB, msH, bullMss ? "MSS" : "BOS", bullCol, true)
        lastBullLn := l
        lastBullLb := t
    trend := 1
    msH := na
if bearBreak
    if showMs
        if not msHist
            line.delete(lastBearLn)
            label.delete(lastBearLb)
        [l, t] = f_ms(msLB, msL, bearMss ? "MSS" : "BOS", bearCol, false)
        lastBearLn := l
        lastBearLb := t
    trend := -1
    msL := na

// ============================================================================
// 7. SETUP COMPLET : SFP → FVG → MSS
// ============================================================================
var int   bSfpBar = na
var float bSfpExt = na
var bool  bFvgOk  = false
var bool  bMssOk  = false
var int   sSfpBar = na
var float sSfpExt = na
var bool  sFvgOk  = false
var bool  sMssOk  = false

bool buySig  = false
bool sellSig = false

// haussier : liquidité basse prise
if bullSfp
    bSfpBar := bar_index
    bSfpExt := low
    bFvgOk  := false
    bMssOk  := false
else if not na(bSfpBar)
    if close < bSfpExt or bar_index - bSfpBar > setWin
        bSfpBar := na
    else
        if bullFvg
            bFvgOk := true
        if bullMss
            bMssOk := true
        if bFvgOk and bMssOk
            buySig  := true
            bSfpBar := na

// baissier : liquidité haute prise
if bearSfp
    sSfpBar := bar_index
    sSfpExt := high
    sFvgOk  := false
    sMssOk  := false
else if not na(sSfpBar)
    if close > sSfpExt or bar_index - sSfpBar > setWin
        sSfpBar := na
    else
        if bearFvg
            sFvgOk := true
        if bearMss
            sMssOk := true
        if sFvgOk and sMssOk
            sellSig := true
            sSfpBar := na

if showSet and buySig
    label.new(bar_index, low, "BUY", style = label.style_label_up, color = bullCol, textcolor = color.white, size = size.small, yloc = yloc.belowbar)
if showSet and sellSig
    label.new(bar_index, high, "SELL", style = label.style_label_down, color = bearCol, textcolor = color.white, size = size.small, yloc = yloc.abovebar)

        string lastTxt = lastSfpDir == 0 ? "—" : (lastSfpDir == 1 ? "▲ " : "▼ ") + (na(lastVolPct) ? "vol. n/d" : str.tostring(lastVolPct, "#.#") + " % hors swing")
        table.cell(dash, 0, 3, "Dernier SFP", text_color = mut, bgcolor = bg, text_size = size.small, text_halign = text.align_left)

// ═══════════════════════════ Trading-day history ═══════════════════════════
// A trading day starts at 18:00 ET. Every drawing is tagged with the last trading day
// it belongs to; when a new day starts, drawings older than `daysBack` days are deleted.
int  sm       = (mins - 1080 + 1440) % 1440       // minutes since 18:00 ET
bool new_tday = sm < nz(sm[1], sm)

var int dayN = 0
if new_tday
    dayN += 1

int  keepFrom = dayN - daysBack
int  KEEP     = 1000000000                        // "keep for now" tag (current PWH/PWL)

f_purgeLines(array<line> a, array<int> t) =>
    if new_tday and a.size() > 0
        for i = a.size() - 1 to 0
            if t.get(i) < keepFrom
                a.get(i).delete()
                a.remove(i)
                t.remove(i)

f_purgeLabels(array<label> a, array<int> t) =>
    if new_tday and a.size() > 0
        for i = a.size() - 1 to 0
            if t.get(i) < keepFrom
                a.get(i).delete()
                a.remove(i)
                t.remove(i)

// Tracks the high/low of a window. When it ends, draws the levels and extends them
// (until 16:00 / a new day if stopAtCutoff, otherwise until the next window ends).
// High/low are tracked even when hidden, so alerts and state never get stuck.
// dayAhead: 1 when the levels are used on the NEXT trading day (PDH/PDL).
// persistent: keep the latest pair no matter how old (PWH/PWL); the previous pair
// ages out from the day it was replaced.
f_range(bool active, bool show, color col, string hiTxt, string loTxt, int width, bool stopAtCutoff, int dayAhead, bool persistent) =>
    var float hi        = na
    var float lo        = na
    var int   t0        = na
    var bool  was       = false
    var line  hiLn      = na
    var line  loLn      = na
    var label hiLb      = na
    var label loLb      = na
    var bool  extending = false
    var array<line>  lns  = array.new<line>()
    var array<int>   lTag = array.new<int>()
    var array<label> lbs  = array.new<label>()
    var array<int>   bTag = array.new<int>()

    f_purgeLines(lns, lTag)
    f_purgeLabels(lbs, bTag)

    bool ended = was and not active
    if active
        hi := was ? math.max(hi, high) : high
        lo := was ? math.min(lo, low) : low
        t0 := was ? t0 : time

    if ended and show
        // The pair being replaced was last shown today
        if persistent and lTag.size() >= 2
            lTag.set(lTag.size() - 1, dayN)
            lTag.set(lTag.size() - 2, dayN)
            bTag.set(bTag.size() - 1, dayN)
            bTag.set(bTag.size() - 2, dayN)
        int tag = persistent ? KEEP : dayN + dayAhead
        hiLn := line.new(t0, hi, time + EXT, hi, xloc = xloc.bar_time, color = col, width = width)
        loLn := line.new(t0, lo, time + EXT, lo, xloc = xloc.bar_time, color = col, width = width)
        hiLb := label.new(time + EXT, hi, hiTxt, xloc = xloc.bar_time, color = color.new(color.black, 100), style = label.style_label_left, textcolor = col, size = LBL)
        loLb := label.new(time + EXT, lo, loTxt, xloc = xloc.bar_time, color = color.new(color.black, 100), style = label.style_label_left, textcolor = col, size = LBL)
        lns.push(hiLn)
        lns.push(loLn)
        lTag.push(tag)
        lTag.push(tag)
        lbs.push(hiLb)
        lbs.push(loLb)
        bTag.push(tag)
        bTag.push(tag)
        extending := true
    else if extending
        if stopAtCutoff and (mins >= CUTOFF or new_day)
            extending := false
        else
            line.set_x2(hiLn, time + EXT)
            line.set_x2(loLn, time + EXT)
            label.set_x(hiLb, time + EXT)
            label.set_x(loLb, time + EXT)

    was := active
    [ended, ended ? hi : na, ended ? lo : na]

// Horizontal line at the open of the first bar at/after minute `t`, extended until 16:00.
f_level(int t, bool show, color col, string txt) =>
    var line  ln   = na
    var label lb   = na
    var bool  live = false
    var array<line>  lns  = array.new<line>()
    var array<int>   lTag = array.new<int>()
    var array<label> lbs  = array.new<label>()
    var array<int>   bTag = array.new<int>()
    f_purgeLines(lns, lTag)
    f_purgeLabels(lbs, bTag)
    bool hit = mins >= t and mins - t < tf_min and (new_day or nz(mins[1], -1) < t)
    if show
        if hit
            ln   := line.new(time, open, time + EXT, open, xloc = xloc.bar_time, color = col, style = line.style_dotted)
            lb   := label.new(time + EXT, open, txt, xloc = xloc.bar_time, color = color.new(color.black, 100), style = label.style_label_left, textcolor = col, size = LBL)
            lns.push(ln)
            lTag.push(dayN)
            lbs.push(lb)
            bTag.push(dayN)
            live := true
        else if live
            if mins >= CUTOFF or new_day
                live := false
            else
                line.set_x2(ln, time + EXT)
                label.set_x(lb, time + EXT)
    live

f_breakUp(float lvl) => not na(lvl) and lvl == lvl[1] and high > lvl and high[1] <= lvl
f_breakDn(float lvl) => not na(lvl) and lvl == lvl[1] and low < lvl and low[1] >= lvl

// One line per trading day for each level, like the other Frame levels: it starts at the day's open (18:00)
// and stops extending at 16:00. Older days are cleared by "Days back to show".
f_hline(float price, bool show, color col, string txt, int width) =>
    var line  ln   = na
    var label lb   = na
    var bool  live = false
    var float last = na
    var array<line>  lns  = array.new<line>()
    var array<int>   lTag = array.new<int>()
    var array<label> lbs  = array.new<label>()
    var array<int>   bTag = array.new<int>()
    f_purgeLines(lns, lTag)
    f_purgeLabels(lbs, bTag)
    if show and not na(price)
        bool fresh = new_tday or na(last) or price != last
        if fresh
            if live                                   // the level changed during the day: close the old line here
                line.set_x2(ln, time)
                label.delete(lb)
            ln   := line.new(time, price, time + EXT, price, xloc = xloc.bar_time, color = col, width = width)
            lb   := label.new(time + EXT, price, txt, xloc = xloc.bar_time, style = label.style_label_left, color = color.new(color.white, 100), textcolor = col, size = LBL)
            lns.push(ln)
            lTag.push(dayN)
            lbs.push(lb)
            bTag.push(dayN)
            last := price
            live := mins < CUTOFF or mins >= 1080
        else if live
            if mins >= CUTOFF and mins < 1080         // 16:00: stop extending
                live := false
            else
                line.set_x2(ln, time + EXT)
                label.set_x(lb, time + EXT)
    live

// One continuous line for the whole period (week or year), from the period's open.
// The week line stops at Friday 16:00; the year line runs until the next year opens.
f_span(float price, int t0, bool show, color col, string txt, int width, bool endFriday) =>
    var line  ln   = na
    var label lb   = na
    var float last = na
    var bool  live = false
    if show and not na(price) and not na(t0)
        if na(last) or price != last                  // a new week / year: replace the old line
            line.delete(ln)
            label.delete(lb)
            ln   := line.new(t0, price, time + EXT, price, xloc = xloc.bar_time, color = col, width = width)
            lb   := label.new(time + EXT, price, txt, xloc = xloc.bar_time, style = label.style_label_left, color = color.new(color.white, 100), textcolor = col, size = LBL)
            last := price
            live := true
        else if live
            if endFriday and et_day == dayofweek.friday and mins >= CUTOFF
                live := false                         // the week ends Friday 16:00
            else
                line.set_x2(ln, time + EXT)
                label.set_x(lb, time + EXT)
    else if not show and not na(ln)
        line.delete(ln)
        label.delete(lb)
        ln   := na
        lb   := na
        last := na
        live := false
    live

f_cxSig() =>
    // CX Press (calcul de CX_Press.mq5)
    float bRaw  = close - math.min(low, nz(close[1], open))
    float sRaw  = math.max(high, nz(close[1], open)) - close
    float vm    = ta.sma(volume, csCandle)
    float rv    = csUseVol and nz(volume) > 0 and nz(vm) > 0 ? math.min(volume / vm, 5.0) : 1.0
    float bE    = ta.ema(bRaw * rv, csCandle)
    float sE    = ta.ema(sRaw * rv, csCandle)
    float bb    = ta.sma(bE, csNorm)
    float sb    = ta.sma(sE, csNorm)
    float bp    = ta.ema(nz(bb) > 0 ? 200.0 * bE / bb : 0.0, csSmooth)
    float sp    = ta.ema(nz(sb) > 0 ? 200.0 * sE / sb : 0.0, csSmooth)
    float nRaw  = bp - sp
    float nsd   = ta.stdev(nRaw, csNorm)
    float nP    = nz(nsd) > 0 ? nRaw / nsd * 300.0 : 0.0
    float fast  = ta.ema(nP, csFastLen)
    float core  = ta.ema(nP, csCoreLen)
    float fm    = ta.sma(fast, csNorm)
    float fsd   = ta.stdev(fast, csNorm)
    int   pStr  = not na(fm) and fast > 0 and core > 0 and fast > fm + csStrongSd * fsd ? 1 : not na(fm) and fast < 0 and core < 0 and fast < fm - csStrongSd * fsd ? -1 : 0
    int   pCore = core > 0 ? 1 : core < 0 ? -1 : 0
    float atr   = ta.atr(14)
    // Croix ✕ (SFP brut de CX Système)
    float pH = ta.pivothigh(high, sfpLen, sfpLen)
    float pL = ta.pivotlow(low, sfpLen, sfpLen)
    var float lvH = na
    var float lvL = na
    if not na(pH)
        lvH := pH
    if not na(pL)
        lvL := pL
    bool rBear = not na(lvH) and high > lvH and close < lvH and open < lvH
    bool rBull = not na(lvL) and low < lvL and close > lvL and open > lvL
    int  sfp   = rBear and rBull ? 0 : rBull ? 1 : rBear ? -1 : 0
    if rBear
        lvH := na
    if rBull
        lvL := na
    if not na(lvH) and close > lvH
        lvH := na
    if not na(lvL) and close < lvL
        lvL := na
    // Carte de liquidité (heure de New York) : jour 18:00, semaine du dimanche 18:00, sessions
    int  nyMin  = hour(time, "America/New_York") * 60 + minute(time, "America/New_York")
    int  tShift = time + 6 * 3600000
    int  dayKey = dayofmonth(tShift, "America/New_York")
    bool newDay = dayKey != nz(dayKey[1], dayKey)
    int  dowS   = dayofweek(tShift, "America/New_York")
    bool newWk  = newDay and dowS == dayofweek.monday
    int  ses    = nyMin >= 1080 or nyMin < 150 ? 0 : nyMin < 420 ? 1 : nyMin < 690 ? 2 : nyMin < 780 ? 3 : nyMin < 960 ? 4 : -1
    var array<float> slP = array.new<float>(14, na)          // 0 PDH 1 PDL 2 PWH 3 PWL 4..13 sessions (haut, bas)
    var array<bool>  slT = array.new<bool>(14, false)
    var float dHi = na
    var float dLo = na
    var float wHi = na
    var float wLo = na
    var float sHi = na
    var float sLo = na
    var int   curS = -2
    if newDay and not na(dHi)
        array.set(slP, 0, dHi), array.set(slT, 0, false)
        array.set(slP, 1, dLo), array.set(slT, 1, false)
    if newDay
        dHi := na
        dLo := na
    if newWk and not na(wHi)
        array.set(slP, 2, wHi), array.set(slT, 2, false)
        array.set(slP, 3, wLo), array.set(slT, 3, false)
    if newWk
        wHi := na
        wLo := na
    if ses != curS
        if curS >= 0 and not na(sHi)
            array.set(slP, 4 + curS * 2, sHi), array.set(slT, 4 + curS * 2, false)
            array.set(slP, 5 + curS * 2, sLo), array.set(slT, 5 + curS * 2, false)
        curS := ses
        sHi := na
        sLo := na
    bool hiTaken = false
    bool loTaken = false

rbUpdateRange(box b, line m, float tp, float bt, int x) =>
    float mid = (tp + bt) / 2
    if not na(b)
        box.set_top(b, tp)
        box.set_bottom(b, bt)
        box.set_right(b, x)
    if not na(m)
        line.set_y1(m, mid)
        line.set_y2(m, mid)
        line.set_x2(m, x)
